cmake_minimum_required(VERSION 3.21)
project(eco_workstation VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ECO_CORE_ONLY "Build only portable engine tests" OFF)
include(CTest)
add_library(light-dsp STATIC light/fm_synth.c light/tb303.c light/drum_rack.c light/mix_scalar.c)
target_include_directories(light-dsp PUBLIC light)
set_property(TARGET light-dsp PROPERTY C_STANDARD 11)
option(LIGHT_ENABLE_NEON "Use LIGHT AArch64 NEON mixing kernels" ON)
if(LIGHT_ENABLE_NEON AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  target_sources(light-dsp PRIVATE light/mix_neon.S)
  target_compile_definitions(light-dsp PRIVATE LIGHT_USE_NEON=1)
endif()
if(UNIX)
  target_link_libraries(light-dsp PUBLIC m)
endif()
add_executable(eco-render src/render.cpp)
target_link_libraries(eco-render PRIVATE light-dsp)
add_executable(eco-bench src/bench.cpp)
target_link_libraries(eco-bench PRIVATE light-dsp)
if(BUILD_TESTING)
  add_executable(eco-engine-tests tests/engine_test.cpp)
  target_include_directories(eco-engine-tests PRIVATE src)
  target_link_libraries(eco-engine-tests PRIVATE light-dsp)
  add_test(NAME engine COMMAND eco-engine-tests)

  add_executable(eco-project-io-tests tests/project_io_test.cpp)
  target_include_directories(eco-project-io-tests PRIVATE src light)
  add_test(NAME project_io COMMAND eco-project-io-tests)
endif()
if(NOT ECO_CORE_ONLY)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
  qt_standard_project_setup()
  qt_add_executable(eco-workstation src/main.cpp src/controller.hpp)
  qt_add_resources(eco-workstation qml PREFIX "/" FILES Main.qml)
  target_link_libraries(eco-workstation PRIVATE light-dsp Qt6::Quick Qt6::QuickControls2 PkgConfig::PIPEWIRE PkgConfig::ALSA Threads::Threads)
  install(TARGETS eco-workstation RUNTIME DESTINATION bin)
endif()
