cmake_minimum_required(VERSION 3.21)
project(eco_workstation VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ECO_CORE_ONLY "Build only portable engine tests" OFF)
include(CTest)
add_library(light-dsp STATIC light/fm_synth.c light/tb303.c light/drum_rack.c light/mix_scalar.c)
target_include_directories(light-dsp PUBLIC light)
set_property(TARGET light-dsp PROPERTY C_STANDARD 11)
option(LIGHT_ENABLE_NEON "Use LIGHT AArch64 NEON mixing kernels" ON)
if(LIGHT_ENABLE_NEON AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  target_sources(light-dsp PRIVATE light/mix_neon.S)
  target_compile_definitions(light-dsp PRIVATE LIGHT_USE_NEON=1)
endif()
if(UNIX)
  target_link_libraries(light-dsp PUBLIC m)
endif()
set(CLAP_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/clap/include)
set(CLAP_HOST_LIBS ${CMAKE_DL_LIBS})
add_executable(eco-render src/render.cpp)
target_include_directories(eco-render PRIVATE ${CLAP_INCLUDE_DIR})
target_link_libraries(eco-render PRIVATE light-dsp ${CLAP_HOST_LIBS})
add_executable(eco-bench src/bench.cpp)
target_include_directories(eco-bench PRIVATE ${CLAP_INCLUDE_DIR})
target_link_libraries(eco-bench PRIVATE light-dsp ${CLAP_HOST_LIBS})
if(BUILD_TESTING)
  add_executable(eco-engine-tests tests/engine_test.cpp)
  target_include_directories(eco-engine-tests PRIVATE src ${CLAP_INCLUDE_DIR})
  target_link_libraries(eco-engine-tests PRIVATE light-dsp ${CLAP_HOST_LIBS})
  add_test(NAME engine COMMAND eco-engine-tests)

  add_executable(eco-project-io-tests tests/project_io_test.cpp)
  target_include_directories(eco-project-io-tests PRIVATE src light)
  add_test(NAME project_io COMMAND eco-project-io-tests)

  # Test-only fixture: a minimal gated-square CLAP instrument, built as a
  # real plugin binary so eco-clap-host-tests exercises ClapHost against an
  # actual dlopen'd CLAP DSO rather than a mock.
  add_library(eco-test-synth-clap MODULE tests/fixtures/test_synth.cpp)
  target_include_directories(eco-test-synth-clap PRIVATE ${CLAP_INCLUDE_DIR})
  set_target_properties(eco-test-synth-clap PROPERTIES PREFIX "" OUTPUT_NAME "eco_test_synth")

  add_executable(eco-clap-host-tests tests/clap_host_test.cpp)
  target_include_directories(eco-clap-host-tests PRIVATE src ${CLAP_INCLUDE_DIR})
  target_link_libraries(eco-clap-host-tests PRIVATE light-dsp ${CLAP_HOST_LIBS})
  target_compile_definitions(eco-clap-host-tests PRIVATE
    TEST_SYNTH_CLAP_PATH="$<TARGET_FILE:eco-test-synth-clap>")
  add_dependencies(eco-clap-host-tests eco-test-synth-clap)
  add_test(NAME clap_host COMMAND eco-clap-host-tests)
endif()
if(NOT ECO_CORE_ONLY)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
  pkg_check_modules(ALSA REQUIRED IMPORTED_TARGET alsa)
  qt_standard_project_setup()
  qt_add_executable(eco-workstation src/main.cpp src/controller.hpp)
  qt_add_resources(eco-workstation qml PREFIX "/" FILES Main.qml)
  target_include_directories(eco-workstation PRIVATE ${CLAP_INCLUDE_DIR})
  target_link_libraries(eco-workstation PRIVATE light-dsp Qt6::Quick Qt6::QuickControls2 PkgConfig::PIPEWIRE PkgConfig::ALSA Threads::Threads ${CLAP_HOST_LIBS})
  install(TARGETS eco-workstation RUNTIME DESTINATION bin)
endif()
